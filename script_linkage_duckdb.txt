----------------------------------------------------------------------------------------------
--- Remove os caracteres *, ", :, e espaços em branco (\s) do início (^) e do final ($) da string.
-----------------------------------------------------------------------------------------------
DROP MACRO IF EXISTS clean_string;

CREATE OR REPLACE MACRO clean_string(input_string) AS 
(
  trim(
    REGEXP_REPLACE(
      REGEXP_REPLACE(
        REGEXP_REPLACE(
          upper(
            TRANSLATE(
              trim(input_string),
              'áàãâäéèêëíìîïóòõôöúùûüçÁÀÃÂÄÉÈÊËÍÌÎÏÓÒÕÔÖÚÙÛÜÇ',
              'aaaaaeeeeiiiiooooouuuucAAAAAEEEEIIIIOOOOOUUUUC'
            )
          ),
           '(^| )+(ALEGADO|IGNORADO|DESCONHECIDO|NAO INFORMADO)[., ]*', ' ', 'g'  
        ),
        '[^A-Z ]+', '', 'g'
      ),
      '\\s{2,}', ' ', 'g'
    )
  )
);

-- Extraindo o primeiro, segundo e terceiro e ultimonome do 'NOME'
UPDATE tb_sim
SET primeironome = SUBSTR(nome, 1, INSTR(nome, ' ') - 1),
    segundonome = CASE
                     WHEN LENGTH(nome) - LENGTH(REPLACE(nome, ' ', '')) >= 1
                     THEN SUBSTR(nome, 
                                 INSTR(nome, ' ') + 1, 
                                 INSTR(SUBSTR(nome, INSTR(nome, ' ') + 1), ' ') - 1)
                     ELSE ''
                  END,
    terceironome = CASE
                      WHEN LENGTH(nome) - LENGTH(REPLACE(nome, ' ', '')) >= 2
                      THEN SUBSTR(
                          SUBSTR(nome, INSTR(nome, ' ') + 1),
                          INSTR(SUBSTR(nome, INSTR(nome, ' ') + 1), ' ') + 1,
                          CASE 
                              WHEN INSTR(SUBSTR(SUBSTR(nome, INSTR(nome, ' ') + 1), INSTR(SUBSTR(nome, INSTR(nome, ' ') + 1), ' ') + 1), ' ') > 0 
                              THEN INSTR(SUBSTR(SUBSTR(nome, INSTR(nome, ' ') + 1), INSTR(SUBSTR(nome, INSTR(nome, ' ') + 1), ' ') + 1), ' ') - 1
                              ELSE LENGTH(SUBSTR(nome, INSTR(nome, ' ') + 1))
                          END
                      )
                      ELSE ''
                   END,
   ultimonome = SUBSTR(nome, LENGTH(nome) - INSTR(REVERSE(nome), ' ') + 2)
WHERE INSTR(nome, ' ') > 0;

-- Adicionar a coluna 'id' como identificador único, se não existir
ALTER TABLE tb_sinan ADD COLUMN IF NOT EXISTS id INTEGER;

-- Preencher a coluna 'id' com uma sequência numérica única
-- Criar uma tabela temporária com a nova estrutura incluindo 'id' e 'hash_deterministico'
CREATE OR REPLACE TEMP TABLE temp_tb_sinan AS
SELECT
    ROW_NUMBER() OVER () AS id, -- Gera um identificador único para cada linha
    *,
    MD5(nome || nomemae || dtnasc) AS hash_deterministico
    -- Combina os campos NOME, NOMEMAE e DTNASC para gerar um hash determinístico
FROM tb_sinan;

-- Substituir a tabela original pela versão atualizada
DROP TABLE tb_sinan;
ALTER TABLE temp_tb_sinan RENAME TO tb_sinan;


/*------------------------------------------------------------------------
-- Deduplicação Determinística (com base no Hash)
------------------------------------------------------------------------*/

-- Adicionar a coluna 'id2' para deduplicação, se não existir
ALTER TABLE tb_sinan
ADD COLUMN IF NOT EXISTS id2 INTEGER;

-- Atualizar a coluna 'id2' com o menor 'id' dentro de cada grupo de hash determinístico
WITH hash_min_id AS (
    SELECT
        hash_deterministico,
        MIN(id) AS id2_min
    FROM tb_sinan
    GROUP BY hash_deterministico
)
UPDATE tb_sinan AS m
SET id2 = h.id2_min
FROM hash_min_id AS h
WHERE m.hash_deterministico = h.hash_deterministico;

-- Criar uma tabela temporária para identificar duplicados com id2 diferentes
CREATE OR REPLACE TEMP TABLE temp_duplicates_deterministic AS
SELECT
    a.id AS id1,
    b.id AS id2,
    a.hash_deterministico
FROM tb_sinan AS a
JOIN tb_sinan AS b
    ON a.hash_deterministico = b.hash_deterministico
WHERE a.id2 <> b.id2  -- Somente registros com id2 diferentes
  AND a.id < b.id     -- Evita comparações duplicadas (a, b) e (b, a)
GROUP BY a.id, b.id, a.hash_deterministico;

-- Substituir id2 NULL pelo próprio id
UPDATE tb_sinan
SET id2 = id
WHERE id2 IS NULL;


/*------------------------------------------------------------------------
-- Deduplicação Probabilística (Deduplição em ciclos)
------------------------------------------------------------------------*/


/*------------------------------------------------------------------------
-- Primeira Passada
------------------------------------------------------------------------*/

-- Criar uma tabela temporária para armazenar os pares duplicados probabilísticos
CREATE OR REPLACE TEMP TABLE temp_duplicates_probabilistic AS
WITH RECURSIVE duplicates AS (
    -- Primeira Iteração: Identificação Inicial de Duplicatas
    SELECT
        LEAST(p1.id2, p2.id2) AS id1,
        GREATEST(p1.id2, p2.id2) AS id2
    FROM tb_sinan p1
    JOIN tb_sinan p2
        ON p1.id < p2.id
        AND p1.id2 <> p2.id2 -- Apenas registros com id2 diferentes
    WHERE p1.dtnasc = p2.dtnasc
      -- AND p1.sexo = p2.sexo
      AND jaro_winkler_similarity(p1.nome, p2.nome) > 0.90
      AND jaro_similarity(p1.primeironome, p2.primeironome) > 0.90
      AND jaro_similarity(p1.segundonome, p2.segundonome) > 0.85
      AND jaro_similarity(p1.terceironome, p2.terceironome) > 0.85
      AND (
          (p1.nomemae IS NOT NULL AND p2.nomemae IS NOT NULL
              AND jaro_similarity(p1.primeironomemae, p2.primeironomemae) > 0.90
              AND jaro_similarity(p1.segundonomemae, p2.segundonomemae) > 0.85)
          OR
          (p1.nomemae IS NULL OR p2.nomemae IS NULL)
      )
    UNION
    SELECT
        LEAST(d.id1, p.id2) AS id1,
        GREATEST(d.id2, p.id2) AS id2
    FROM duplicates d
    JOIN tb_sinan p
        ON d.id2 = p.id2
)
SELECT DISTINCT id1, id2
FROM duplicates;

-- Atualizar id2 na tabela com os pares probabilísticos identificados
UPDATE tb_sinan AS m
SET id2 = t.id1
FROM temp_duplicates_probabilistic AS t
WHERE m.id = t.id2;

-- Substituir id2 NULL pelo próprio id
UPDATE tb_sinan
SET id2 = id
WHERE id2 IS NULL;

-- Segunda Passada Probabilística: Comparar apenas registros com id2 diferentes
CREATE OR REPLACE TEMP TABLE temp_duplicates_probabilistic_second_pass AS
WITH RECURSIVE duplicates AS (
    -- Primeira Iteração: Identificação Inicial de Duplicatas
    SELECT
        LEAST(p1.id2, p2.id2) AS id1,
        GREATEST(p1.id2, p2.id2) AS id2
    FROM tb_sinan p1
    JOIN tb_sinan p2
        ON p1.id < p2.id
        AND p1.id2 <> p2.id2 -- Somente registros com id2 diferentes
    WHERE p1.dtnasc = p2.dtnasc
      -- AND p1.sexo = p2.sexo
      AND jaro_winkler_similarity(p1.nome, p2.nome) > 0.90
      AND jaro_similarity(p1.primeironome, p2.primeironome) > 0.90
      AND jaro_similarity(p1.segundonome, p2.segundonome) > 0.85
      AND jaro_similarity(p1.terceironome, p2.terceironome) > 0.85
      AND (
          (p1.nomemae IS NOT NULL AND p2.nomemae IS NOT NULL
              AND jaro_similarity(p1.primeironomemae, p2.primeironomemae) > 0.90
              AND jaro_similarity(p1.segundonomemae, p2.segundonomemae) > 0.85)
          OR
          (p1.nomemae IS NULL OR p2.nomemae IS NULL)
      )
    UNION
    -- Iterações Recursivas
    SELECT
        LEAST(d.id1, p.id2) AS id1,
        GREATEST(d.id2, p.id2) AS id2
    FROM duplicates d
    JOIN tb_sinan p
        ON d.id2 = p.id2
)
-- Seleção Final
SELECT DISTINCT id1, id2
FROM duplicates;

-- Atualizar a coluna id2 na tabela master_final com os resultados da segunda passada
UPDATE tb_sinan AS m
SET id2 = t.id1
FROM temp_duplicates_probabilistic_second_pass AS t
WHERE m.id = t.id2
  AND m.id2 <> t.id1; -- Atualiza apenas registros com id2 diferentes

-- Substituir id2 NULL pelo próprio id (caso algum registro não seja duplicado após a segunda passada)
UPDATE tb_sinan
SET id2 = id
WHERE id2 IS NULL;

-- Terceira Passada Probabilística com Critérios Adicionais
CREATE OR REPLACE TEMP TABLE temp_duplicates_third_pass AS
WITH RECURSIVE duplicates AS (
    -- Primeira Iteração: Identificação Inicial de Duplicatas
    SELECT
        LEAST(p1.id2, p2.id2) AS id1,
        GREATEST(p1.id2, p2.id2) AS id2
    FROM tb_sinan p1
    JOIN tb_sinan p2
        ON p1.id < p2.id
        AND p1.id2 <> p2.id2 -- Apenas registros com id2 diferentes
    WHERE p1.dtnasc = p2.dtnasc -- Mesma data de nascimento
      -- AND p1.SEXO = p2.SEXO -- Mesmo sexo
      AND jaro_winkler_similarity(p1.nome, p2.nome) > 0.90 -- Similaridade pelo nome completo
      AND jaro_winkler_similarity(p1.primeironome, p2.primeironome) > 0.90 -- Primeiro nome
      AND jaro_winkler_similarity(p1.segundonome, p2.segundonome) > 0.85 -- Segundo nome
      AND jaro_winkler_similarity(p1.terceironome, p2.terceironome) > 0.85 -- Terceiro nome
      AND (
          (p1.nomemae IS NOT NULL AND p2.nomemae IS NOT NULL -- Se o nome da mãe não for nulo
              AND jaro_winkler_similarity(p1.primeironomemae, p2.primeironomemae) > 0.90 -- Primeiro nome da mãe
              AND jaro_winkler_similarity(p1.segundonomemae, p2.segundonomemae) > 0.85) -- Segundo nome da mãe
          OR
          (p1.nomemae IS NULL OR p2.nomemae IS NULL) -- Se o nome da mãe for nulo, ignora
      )
    UNION
    -- Iterações Recursivas
    SELECT
        LEAST(d.id1, p.id2) AS id1,
        GREATEST(d.id2, p.id2) AS id2
    FROM duplicates d
    JOIN tb_sinan p
        ON d.id2 = p.id2
)
-- Seleção Final
SELECT DISTINCT id1, id2
FROM duplicates;

-- Atualizar a coluna id2 na tabela master_final com os resultados da terceira passada
UPDATE tb_sinan AS m
SET id2 = t.id1
FROM temp_duplicates_third_pass AS t
WHERE m.id = t.id2
  AND m.id2 <> t.id1; -- Atualiza apenas registros com id2 diferentes

WHERE id2 IS NULL;

-- Exportando tb_linkage_final
-------------------
-- Mantem apenas o melhor par por notificacao: menor fase = criterio mais
-- rigido (link1 deterministico ... link15 mais frouxo); empate resolvido
-- pelo menor numerodo para o resultado ser deterministico
CREATE OR REPLACE TABLE tb_linkage_1n AS
SELECT id_sinan, nunotific, numerodo, fase
FROM (
    SELECT
        id_sinan, nunotific, numerodo, fase,
        ROW_NUMBER() OVER (
            PARTITION BY nunotific
            ORDER BY fase, numerodo
        ) AS rn
    FROM tb_linkage_dedup
)
WHERE rn = 1;

-- Garante o 1:1 tambem do lado do obito: o passo acima so evita que uma
-- notificacao SINAN tenha mais de um obito pareado, mas nao impede que o
-- MESMO obito seja "o melhor par" de notificacoes diferentes (pessoas
-- diferentes reivindicando o mesmo atestado de obito). Aqui, para cada
-- numerodo, so o par de fase mais rigida sobrevive; os demais ficam sem
-- par nesta rodada (nao registrado, nao um par errado).
CREATE OR REPLACE TABLE tb_linkage_final AS
SELECT id_sinan, nunotific, numerodo
FROM (
    SELECT
        id_sinan, nunotific, numerodo,
        ROW_NUMBER() OVER (
            PARTITION BY numerodo
            ORDER BY fase, nunotific
        ) AS rn2
    FROM tb_linkage_1n
)
WHERE rn2 = 1
ORDER BY
    id_sinan, numerodo;
